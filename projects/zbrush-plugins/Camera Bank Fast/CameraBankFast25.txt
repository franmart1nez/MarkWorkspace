    [RoutineCall,FBF_UsedCameras]
    [Loop,2048,[VarSet,fxHead(fxI),-1],fxI]
    [Loop,[SubToolGetCount],
        [VarSet,fxID,[SubToolGetID,fxI]]
        [VarSet,fxHigh(fxI),INT(fxID/65536)]
        [VarSet,fxLow(fxI),fxID-(fxHigh(fxI)*65536)]
        [VarSet,fxB,16+(fxI*5)]

        [VarSet,fxBucket,(fxLow(fxI)+(fxHigh(fxI)*17))&2047]
        [VarSet,fxPeer,fxHead(fxBucket)]
        [Loop,fxI,
            [If,fxPeer<0,[LoopExit]]
            [If,(fxLow(fxPeer)==fxLow(fxI))&&(fxHigh(fxPeer)==fxHigh(fxI)),
                [VarSet,fxG,fbTrackGroups(fxI)-1]
                [VarSet,fxP,fbTrackGroups(fxPeer)-1]
                [If,fxG!=fxP,
                    [MVarSet,FBFCache,fxB+4,1][MVarSet,FBFCache,16+(fxPeer*5)+4,1]
                    [If,(([MVarGet,FB25Data,25746+(fxG*2)]&fxUsedLow)!=([MVarGet,FB25Data,25746+(fxP*2)]&fxUsedLow))||(([MVarGet,FB25Data,25747+(fxG*2)]&fxUsedHigh)!=([MVarGet,FB25Data,25747+(fxP*2)]&fxUsedHigh)),

[RoutineDef,FBF_TryCache,
    [VarSet,fxCacheHit,0]
    [If,[MemGetSize,FBFCache]>0,
        [If,[MVarGet,FBFCache,0]==1,
            [If,([MVarGet,FBFCache,1]==[SubToolGetCount])&&([MVarGet,FBFCache,2]==[ToolGetActiveIndex]),
                [If,([MVarGet,FBFCache,3]==[MVarGet,FB25Data,8])&&([MVarGet,FBFCache,4]==[MVarGet,FB25Data,9]),
                    [If,[MVarGet,FBFCache,5]==[MVarGet,FB25Data,25745],[VarSet,fxCacheHit,1]]
                ]
            ]
        ]
    ]
    [If,fxCacheHit==1,
        [Loop,[SubToolGetCount],
            [VarSet,fxB,16+(fxI*5)]
            [VarSet,fxID,[SubToolGetID,fxI]][VarSet,fxHi,INT(fxID/65536)]
            [VarSet,fxLo,fxID-(fxHi*65536)]
            [If,(fxLo!=[MVarGet,FBFCache,fxB])||(fxHi!=[MVarGet,FBFCache,fxB+1]),[VarSet,fxCacheHit,0][LoopExit]]
            [VarSet,fbTrackGroups(fxI),[MVarGet,FBFCache,fxB+2]]
        ,fxI]
    ]
    [If,fxCacheHit==1,
        [VarSet,fxSelected,[SubToolGetActiveIndex]]
        [VarSet,fbTrackPrefix,[StrMerge,"CB",[MVarGet,FB25Data,8],[MVarGet,FB25Data,9],"G"]]
        [RoutineCall,FB32_ReadName][RoutineCall,FB32_ParseTag,fbTrackName]
        [If,fbParsedGroup!=fbTrackGroups(fxSelected),
            [VarSet,fxCacheHit,0]
            // Preserve the reviewed first-SubTool filename alias exception.
            [If,(fxSelected==0)&&(fbTagCount==0),
                [VarSet,fxTool,[ToolGetActiveIndex]][VarSet,fxPath,[ToolGetPath,fxTool]]
                [VarSet,fxAlias,[FileNameExtract,fxPath,6]]
                [If,([StrLength,fxAlias]>0)&&([StrLength,fxAlias]==[StrLength,fbTrackName]),

                        [If,fbFirstIdGroup==fbTrackGroups(0),[VarSet,fxCacheHit,1]]
                    ]
                ]
            ]
        ]
    ]
    [If,fxCacheHit==1,
        [VarSet,fxNeedsSelection,0]
        [Loop,[SubToolGetCount],
            [If,[MVarGet,FBFCache,16+(fxI*5)+3+fxStoreCheck]==1,[VarSet,fxNeedsSelection,1][LoopExit]]
        ,fxI]
        [If,fxNeedsSelection==1,
            [If,[MemGetSize,FBFReadView]>0,[MemDelete,FBFReadView]]

            [MVarSet,FBFReadView,9,[CanvasZoomGet]]
            [MVarSet,FBFReadView,10,[CanvasPanGetH]][MVarSet,FBFReadView,11,[CanvasPanGetV]]
            [IFreeze,[Loop,[SubToolGetCount],
                [If,[MVarGet,FBFCache,16+(fxI*5)+3+fxStoreCheck]==1,
                    [SubToolSelect,fxI][RoutineCall,FB32_ReadName][RoutineCall,FB32_ParseTag,fbTrackName]
                    [If,fbParsedGroup!=fbTrackGroups(fxI),[VarSet,fxCacheHit,0][LoopExit]]
                ]
            ,fxI]]
            [SubToolSelect,fxSelected][MTransformSet,FBFReadView,0]
            [CanvasZoomSet,[MVarGet,FBFReadView,9]][CanvasPanSet,[MVarGet,FBFReadView,10],[MVarGet,FBFReadView,11]]
            [MemDelete,FBFReadView]
        ]
    ]

// Native ZAppLink .vws: eight records of ten float32 cells, little endian.
// Order: Front, Back, Rght, Left, Top, Botm, Cust1, Cust2. Cell 9 is occupied.
// No external executable, native ZAppLink replacement or filename lens guessing.
[VarDef,vwsSources(8),0]
[VarDef,vwsSlots(8),0]
[VarDef,vwsImported,0]
[RoutineDef,FBV_Reject,
    [If,[MemGetSize,FBVInput]>0,[MemDelete,FBVInput]]
    [MessageOK,vwsError,"Camera Bank Fast"][Exit]
,vwsError]
[RoutineDef,FBV_AcceptVWS,
    [VarSet,vwsImported,0]
    [If,[MemGetSize,FBVInput]!=320,
        [RoutineCall,FBV_Reject,"Unsupported VWS layout. Expected the native 320-byte ZAppLink views file. Current cameras were kept."]
    ]
    [VarSet,vwsCount,0]
    [Loop,8,
        [VarSet,vwsFlag,[MVarGet,FBVInput,(vwsI*10)+9]]
        [If,(vwsFlag!=0)&&(vwsFlag!=1),
            [RoutineCall,FBV_Reject,"Invalid VWS view flag. Current cameras were kept."]
        ]
        [If,vwsFlag==1,
            [Loop,9,
                [VarSet,vwsValue,[MVarGet,FBVInput,(vwsI*10)+vwsJ]]
                [If,(vwsValue>=-100000000)&&(vwsValue<=100000000),,
                    [RoutineCall,FBV_Reject,"Invalid VWS transform (non-finite or out of range). Current cameras were kept."]
                ]
                [If,(vwsJ>=3)&&(vwsJ<=5),
                    [If,vwsValue<=0,[RoutineCall,FBV_Reject,"Invalid VWS view scale. Current cameras were kept."]]
                ]
            ,vwsJ]
            [VarSet,vwsSources(vwsCount),vwsI][VarInc,vwsCount]
        ]
    ,vwsI]
    [If,vwsCount==0,[RoutineCall,FBV_Reject,"This VWS file contains no stored views. Current cameras were kept."]]
    [RoutineCall,FB31_Require3D]
    [If,[SubToolGetCount]>1000,[RoutineCall,FBV_Reject,"Camera Bank Fast supports up to 1000 SubTools."]]
    [RoutineCall,FB32_Recover]
    [VarSet,vwsFree,0][VarSet,vwsOccupied,0]
    [Loop,25,
        [If,[MVarGet,FB25Data,20+(vwsI*20)]==1,
            [VarInc,vwsOccupied]
        ,
            [If,vwsFree<vwsCount,[VarSet,vwsSlots(vwsFree),vwsI][VarInc,vwsFree]]
        ]
    ,vwsI]
    [If,vwsFree<vwsCount,[RoutineCall,FBV_Reject,"Not enough free camera slots for this entire VWS set. Export your bank or free slots, then retry. No cameras were replaced."]]
    [If,(vwsOccupied>0)&&([MVarGet,FB25Data,7]!=1),
        [RoutineCall,FBV_Reject,"Review and UPDATE ACTIVE for every untracked legacy camera before adding VWS views, or start an empty bank."]
    ]
    [VarSet,vwsActive,[MVarGet,FB25Data,5]]
    [VarSet,currentSubCount,[SubToolGetCount]]
    [VarSet,fxStoreCheck,1][VarSet,fxWriteFast,0]
    [If,[MVarGet,FB25Data,7]==1,
        [RoutineCall,FB32_RebaseForEdit]
    ,
        [RoutineCall,FB32_BeginTransaction]
    ]
    // One existing recoverable writer transaction covers the complete batch,
    // including initial model binding. No native view or document is changed.
    [Loop,vwsCount,
        [RoutineCall,FBV_CaptureSlot,vwsSlots(vwsI),1]
        [VarSet,base,20+(vwsSlots(vwsI)*20)]
        [MemCopy,FBVInput,vwsSources(vwsI)*40,FB25Data,(base+1)*4,36]
        [MVarSet,FB25Data,base+1,[MVarGet,FB25Data,base+1]*[IGet,"Document:Width"]]
        [MVarSet,FB25Data,base+2,[MVarGet,FB25Data,base+2]*[IGet,"Document:Height"]]
        [Loop,3,
            [MVarSet,FB25Data,base+4+vwsJ,[MVarGet,FB25Data,base+4+vwsJ]*[IGet,"Document:Width"]]
        ,vwsJ]
    ,vwsI]
    [MVarSet,FB25Data,5,vwsActive]
    [RoutineCall,FB32_FinishStore]
    [VarSet,fxStoreCheck,0]
    [VarSet,vwsImported,vwsCount]
    [MemDelete,FBVInput]
    [RoutineCall,FB29_SyncVisuals]
]
[RoutineDef,FBV_ImportVWS,
    [VarSet,vwsImported,0]
    [VarSet,vwsFile,[FileNameAsk,"ZAppLink Views (*.vws)|*.vws||",,"Import native ZBrush views"]]
    [If,[StrLength,vwsFile]==0,[Exit]]
    [If,[MemGetSize,FBVInput]>0,[MemDelete,FBVInput]]
    [MemCreateFromFile,FBVInput,vwsFile]
    [RoutineCall,FBV_AcceptVWS]
    [MessageOK,[StrMerge,vwsImported," VWS views added to free slots with Render ON. Current canvas, lens and visibility were used. To adjust a view, recall it and press UPDATE ACTIVE after editing. Save the matching ZTL/ZPR and EXPORT BANK."],"Camera Bank Fast"]
]

[RoutineDef,FB321_ResolveFirstID,
    [VarSet,fbFirstIdGroup,0]
    [If,[MemGetSize,FB25Data]>=123088,
        [VarSet,fbIdRecords,[MVarGet,FB25Data,27771]]
        [VarSet,fbTargetId,[SubToolGetID,0]]
        [VarSet,fbTargetHi,INT(fbTargetId/65536)]
        [VarSet,fbTargetLo,fbTargetId-(fbTargetHi*65536)]
        [VarSet,fbIdOK,1]
        [Loop,1000,[VarSet,fbIdRemaining(fbIdI),0],fbIdI]
        [Loop,fbIdRecords,
            [If,([MVarGet,FB25Data,27772+(fbIdI*3)]==fbTargetLo)&&([MVarGet,FB25Data,27773+(fbIdI*3)]==fbTargetHi),
                [VarSet,fbIdGroup,[MVarGet,FB25Data,27774+(fbIdI*3)]-1]

            ]

        [Loop,[SubToolGetCount]-1,
            [VarSet,fbIdPeer,fbIdI+1]
            [If,[SubToolGetID,fbIdPeer]==fbTargetId,
                [VarSet,fbIdGroup,fbTrackGroups(fbIdPeer)-1]
                [VarSet,fbIdRemaining(fbIdGroup),fbIdRemaining(fbIdGroup)-1]
                [If,fbIdRemaining(fbIdGroup)<0,[VarSet,fbIdOK,0]]
            ]
        ,fbIdI]
        [Loop,[MVarGet,FB25Data,25745],
            [If,fbIdRemaining(fbIdI)>0,
                [If,fbFirstIdGroup>0,[VarSet,fbIdOK,0]]
                [VarSet,fbFirstIdGroup,fbIdI+1]

    ]
