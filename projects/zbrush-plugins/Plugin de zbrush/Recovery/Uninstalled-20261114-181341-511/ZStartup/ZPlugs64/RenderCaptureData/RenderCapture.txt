[RoutineDef,RC_CB29_RestoreVisualState,
    [VarSet,visBase,
        520 + (slot * 1001)
    ]
    [VarSet,savedCount,
        [MVarGet,CB25Data,visBase]
    ]
    [VarSet,currentCount,
        [SubToolGetCount]
    ]
    [If,
        (savedCount > 0)
        &&
        (savedCount == currentCount),
        [VarSet,st,0]
        [Loop,savedCount,
            [VarSet,savedState,
                [MVarGet,
                    CB25Data,
                    visBase + 1 + st
                ]
            ]
            [VarSet,savedEye,
                (savedState & 0x01)
            ]
            [VarSet,currentStatus,
                [SubToolGetStatus,st]
            ]
            [VarSet,currentEye,
                (currentStatus & 0x01)
            ]
            [SubToolSetStatus,
                st,
                (currentStatus - currentEye) + savedEye
            ]
            [VarInc,st]
        ]
        [VarSet,st,0]
        [Loop,savedCount,
            [VarSet,savedState,
                [MVarGet,
                    CB25Data,
                    visBase + 1 + st
                ]
            ]
            [VarSet,savedFolderEye,
                (savedState & 0x02)
            ]
            [VarSet,currentStatus,
                [SubToolGetStatus,st]
            ]
            [VarSet,currentFolderEye,
                (currentStatus & 0x02)
            ]
            [SubToolSetStatus,
                st,
                (currentStatus - currentFolderEye) + savedFolderEye
            ]
            [VarInc,st]
        ]
        [IUpdate,1,1]
    ]
,slot]

    [RoutineCall,RC_Folder,[StrMerge,rcRoot,"/",rcLabel]]
    [VarSet,rcOutput,[StrMerge,rcRoot,"/",rcLabel,"/",rcPhase]]
    [RoutineCall,RC_Folder,rcOutput]
]
// Reject collisions BEFORE changing display state. Create a new session to rerun.
[RoutineDef,RC_CheckOutput,
    [Loop,25,
        [VarSet,rcBase,20+(rcSlot*20)]
        [If,([MVarGet,CB25Data,rcBase]==1)&&([MVarGet,CB25Data,rcBase+16]==1),
            [RoutineCall,RC_OutputFolder]
            [If,rcCheckScreenshots==1,
                [If,[FileExists,[StrMerge,rcRoot,"/Screenshots/",rcLabel,".png"]]>0,
                    [RoutineCall,RC_Fail,"A screenshot already exists for a selected camera. Press CREATE FOLDERS to start a new session; existing images are preserved."]
                ]
            ]
            [Loop,rcPassCount,
                [If,[FileExists,[StrMerge,rcOutput,"/",rcPassNames(rcFileIndex)]]>0,
                    [RoutineCall,RC_Fail,"This phase already has output files. Press CREATE FOLDERS to start a new session; existing renders are preserved."]
                ]
            ,rcFileIndex]
        ]
    ,rcSlot]
,rcCheckScreenshots]
[RoutineDef,RC_Export,
    [FileNameSetNext,[Var,rcOutputFile]]
    [IPress,[Var,rcExportID]]
    [IUpdate,1,1]
    [If,[FileExists,[Var,rcOutputFile]]==0,[RoutineCall,RC_Fail,[StrMerge,"Export failed: ",rcOutputFile]]]
,rcExportID,rcOutputFile]
[RoutineDef,RC_CopyScreenshot,
    [VarSet,rcShotFile,[StrMerge,rcRoot,"/Screenshots/",rcLabel,".png"]]
    // FileCopyTo copies the exported PNG bytes; do not render or export again.
    [If,[FileExists,rcShotFile]>0,[RoutineCall,RC_Fail,"Screenshot already exists. Create a new session; existing images are preserved."]]
    [If,[MemGetSize,RC31CopyPath]>0,[MemDelete,RC31CopyPath]]
    [MemCreate,RC31CopyPath,1024,0]
    [MemWriteString,RC31CopyPath,rcShotFile,0]
    [VarSet,rcCopyResult,[FileExecute,[Var,rcDll],"FileCopyTo",[StrMerge,rcOutput,"/Z document.png"],,RC31CopyPath]]
    [MemDelete,RC31CopyPath]
    [If,(rcCopyResult!=0)||([FileExists,rcShotFile]==0),
        [RoutineCall,RC_Fail,[StrMerge,"Screenshot copy failed: ",rcShotFile]]
    ]
]
[RoutineDef,RC_NonBPR,
    [RoutineCall,RC_Restore]
    [RoutineCall,RC_Preflight]
    [RoutineCall,RC_GetRoot]
    [RoutineCall,RC_Materials]
    [RoutineCall,RC_Folder,[StrMerge,rcRoot,"/Screenshots"]]
    [VarSet,rcPhase,"NonBPR"][VarSet,rcPassCount,4]
    [VarSet,rcPassNames(0),"Z document.png"]
    [VarSet,rcPassNames(1),"Normal.png"]
    [VarSet,rcPassNames(2),"Double.png"]
    [VarSet,rcPassNames(3),"Metal.png"]
    [RoutineCall,RC_CheckOutput,1]
    [RoutineCall,RC_Snapshot]
    [Loop,25,
        [VarSet,rcBase,20+(rcSlot*20)]
        [If,([MVarGet,CB25Data,rcBase]==1)&&([MVarGet,CB25Data,rcBase+16]==1),
            [RoutineCall,RC_CB29_LoadSlot,rcSlot]
            [RoutineCall,RC_OutputFolder]
            [ISet,"Render:Render Properties:Shadows",0]
            [IColorSet,255,255,255]
            // Preserve the original workflow: document uses Polypaint ON;
            // Normal / Double / Metal use Polypaint and texture display OFF.
            [RoutineCall,RC_Polypaint,1]
            [IPress,[Var,rcThor]]
            [IPress,"Render:Preview"][IUpdate,2,1]
            [NoteBar,[StrMerge,"Non BPR | Camera ",rcLabel],rcSlot/25]
            [RoutineCall,RC_Export,"Document:Export",[StrMerge,rcOutput,"/",rcPassNames(0)]]
            [RoutineCall,RC_CopyScreenshot]
            [RoutineCall,RC_Polypaint,0]
            [IPress,[Var,rcNormal]]
            [IPress,"Render:Preview"][IUpdate,2,1]
            [RoutineCall,RC_Export,"Document:Export",[StrMerge,rcOutput,"/",rcPassNames(1)]]
            [IPress,[Var,rcDouble]]
            [IPress,"Render:Preview"][IUpdate,2,1]
            [RoutineCall,RC_Export,"Document:Export",[StrMerge,rcOutput,"/",rcPassNames(2)]]
            [IPress,[Var,rcMetal]]
            [IPress,"Render:Preview"][IUpdate,2,1]
            [RoutineCall,RC_Export,"Document:Export",[StrMerge,rcOutput,"/",rcPassNames(3)]]
        ]
    ,rcSlot]
    [RoutineCall,RC_Restore]
    [NoteBar,"Non BPR complete",1]
    [MessageOK,[StrMerge,rcCameraCount," cameras: 4 PNG files + 1 Screenshots copy each. Output: ",rcRoot],"Render Capture"]
]
// Some ZBrush versions cannot address pass thumbnails by their recorded names.
// Numeric IDs are the documented ZBrushCentral fallback; validate their parent.
[RoutineDef,RC_ResolvePass,
    [VarSet,rcPassID,0]
    [If,[IExists,[Var,rcPassPath]],[VarSet,rcPassID,[IGetID,[Var,rcPassPath]]]]
    [If,rcPassID==0,
        [If,[IExists,[Var,rcFallback]],
            [VarSet,rcTitle,[StrLower,[IGetTitle,[Var,rcFallback],1]]]
            [If,([StrFind,"render",rcTitle]>=0)&&([StrFind,"bpr",rcTitle]>=0),[VarSet,rcPassID,rcFallback]]
        ]
    ]
    [If,rcPassID==0,[RoutineCall,RC_Fail,[StrMerge,"Cannot locate BPR pass: ",rcPassPath]]]
    [VarSet,rcPassIDs(rcPassIndex),rcPassID]
,rcPassIndex,rcPassPath,rcFallback]
[RoutineDef,RC_BPR,
    [RoutineCall,RC_Restore]
    [RoutineCall,RC_Preflight]
    [RoutineCall,RC_GetRoot]
    [VarSet,rcPhase,"BPR"][VarSet,rcPassCount,7]
    [VarSet,rcPassNames(0),"_AO.psd"]
    [VarSet,rcPassNames(1),"_Composite.psd"]
    [VarSet,rcPassNames(2),"_Depth.psd"]
    [VarSet,rcPassNames(3),"_Mask.psd"]
    [VarSet,rcPassNames(4),"_Render.psd"]
    [VarSet,rcPassNames(5),"_Shadow.psd"]
    [VarSet,rcPassNames(6),"_SSS.psd"]
    [RoutineCall,RC_ResolvePass,0,"Render:BPR RenderPass:AmOc",2210]
    [RoutineCall,RC_ResolvePass,1,"Render:BPR RenderPass:Com",2206]
    [RoutineCall,RC_ResolvePass,2,"Render:BPR RenderPass:Dep",2208]
    [RoutineCall,RC_ResolvePass,3,"Render:BPR RenderPass:Mask",2211]
    [RoutineCall,RC_ResolvePass,4,"Render:BPR RenderPass:Img",2207]
    [RoutineCall,RC_ResolvePass,5,"Render:BPR RenderPass:Shdw",2209]
    [RoutineCall,RC_ResolvePass,6,"Render:BPR RenderPass:Sss",2315]
    [RoutineCall,RC_CheckOutput,0]
    [RoutineCall,RC_Snapshot]
    [IPress,"Render:Preview"]
    [RoutineCall,RC_Polypaint,1]
    [RoutineCall,RC_PrepareSSSLight]
    [Loop,25,
        [VarSet,rcBase,20+(rcSlot*20)]
        [If,([MVarGet,CB25Data,rcBase]==1)&&([MVarGet,CB25Data,rcBase+16]==1),
            [RoutineCall,RC_CB29_LoadSlot,rcSlot]
            [RoutineCall,RC_OutputFolder]
            [ISet,"Render:Render Properties:Shadows",1]
            [ISet,"Render:Render Properties:AOcclusion",1]
            [ISet,"Render:Render Properties:Sss",1]
            [ISet,"Render:BPR RenderPass:Reuse Existing Maps",0]
            [NoteBar,[StrMerge,"BPR | Camera ",rcLabel],rcSlot/25]
            [ISet,"Render:BPR Shadow:Rays",30]
            [ISet,"Render:BPR Shadow:Angle",15]
            [ISet,"Render:BPR AO:Rays",30]
            [ISet,"Render:BPR RenderPass:SPix",2]
            [IPress,"Render:BPR RenderPass:Render Best Preview"]
            [IUpdate,2,1]
            // Check ALL seven generated maps before exporting any for this view.
            [Loop,7,
                [If,[StrFind,"width=",[StrLower,[IGetInfo,rcPassIDs(rcPass)]]]<0,
                    [RoutineCall,RC_Fail,[StrMerge,"BPR map unavailable: ",rcPassNames(rcPass),". Render may have been cancelled."]]
                ]
            ,rcPass]
            [Loop,7,
                [RoutineCall,RC_Export,rcPassIDs(rcPass),[StrMerge,rcOutput,"/",rcPassNames(rcPass)]]
            ,rcPass]
        ]
    ,rcSlot]
    [RoutineCall,RC_Restore]
    [NoteBar,"BPR complete",1]
    [MessageOK,[StrMerge,rcCameraCount," cameras: 7 PSD files each. Output: ",rcRoot],"Render Capture"]
]
