[RoutineDef,CB29_RestoreVisualState,
    [VarSet,visBase,
        520 + (slot * 1001)
    ]
    [VarSet,savedCount,
        [MVarGet,CB25Data,visBase]
    ]
    [VarSet,currentCount,
        [SubToolGetCount]
    ]
    [If,
        (savedCount > 0)
        &&
        (savedCount == currentCount),
        [VarSet,st,0]
        [Loop,savedCount,
            [VarSet,savedState,
                [MVarGet,
                    CB25Data,
                    visBase + 1 + st
                ]
            ]
            [VarSet,savedEye,
                (savedState & 0x01)
            ]
            [VarSet,currentStatus,
                [SubToolGetStatus,st]
            ]
            [VarSet,currentEye,
                (currentStatus & 0x01)
            ]
            [SubToolSetStatus,
                st,
                (currentStatus - currentEye) + savedEye
            ]
            [VarInc,st]
        ]
        [VarSet,st,0]
        [Loop,savedCount,
            [VarSet,savedState,
                [MVarGet,
                    CB25Data,
                    visBase + 1 + st
                ]
            ]
            [VarSet,savedFolderEye,
                (savedState & 0x02)
            ]
            [VarSet,currentStatus,
                [SubToolGetStatus,st]
            ]
            [VarSet,currentFolderEye,
                (currentStatus & 0x02)
            ]
            [SubToolSetStatus,
                st,
                (currentStatus - currentFolderEye) + savedFolderEye
            ]
            [VarInc,st]
        ]
        [IUpdate,1,1]
    ]
,slot]
