// ID32Recovery: 0 phase (0.5 partial snapshot, 1 display, 2 backup, 3 mutation), 1 selected,
// 2 count, 3 Tool index, 4 material, 5..7 RGB, 8..10 M/RGB/MRGB,
// 11 width,12 height,13 Pro,14 zoom,15/16 pan,17 Solo,18 active camera,
// 19 PolyFrame,20 render mode,21 RGB intensity,22 LightBox,23 Thumbnail,
// 24 restore in progress,25 completed snapshot entries; 32..40 transform;
// 48..59 optional lens fields;
// 64+i*6: ID, status, Colorize, texture toggle (-1 absent), union, reserved.
// ID32Names uses 256-byte slots. Full titles >255 bytes are rejected.
[RoutineDef,ID_ReadPaths,
    [VarSet,idBackup,""][VarSet,idRecord,""]
    [If,[MemGetSize,ID32BackupPath]>0,[MemReadString,ID32BackupPath,idBackup,0,1]]
    [If,[MemGetSize,ID32RecordPath]>0,[MemReadString,ID32RecordPath,idRecord,0,1]]
]
[RoutineDef,ID_ShowBackup,
    // The admitted backup path is <=240: this message is <=251 characters.
    [If,[StrLength,idBackup]>0,
        [MessageOK,[StrMerge,"Pre-ID ZTL: ",idBackup],"Generar y exportar ID"]
    ]
]
[RoutineDef,ID_CheckIdentity,
    [VarSet,idSameTool,1]
    [If,([SubToolGetCount]!=[MVarGet,ID32Recovery,2])||([ToolGetActiveIndex]!=[MVarGet,ID32Recovery,3]),[VarSet,idSameTool,0]]
    [If,idSameTool==1,
        [VarSet,idIdentitySelected,[SubToolGetActiveIndex]]
        [VarSet,idIdentityCount,[SubToolGetCount]]
        [If,[MVarGet,ID32Recovery,0]<1,[VarSet,idIdentityCount,[MVarGet,ID32Recovery,25]]]
        [Loop,idIdentityCount,
            [VarSet,idSavedID,0]
            [MemRead,ID32Recovery,idSavedID,6,(64+(idIdentityIndex*6))*4]
            [If,[SubToolGetID,idIdentityIndex]!=idSavedID,[VarSet,idSameTool,0][LoopExit]]
            [SubToolSelect,idIdentityIndex]
            [VarSet,idSavedTitle,""]
            [MemReadString,ID32Names,idSavedTitle,idIdentityIndex*256]
            [VarSet,idCurrentTitle,[IGetTitle,"Tool:ItemInfo"]]
            [If,([StrLength,idCurrentTitle]!=[StrLength,idSavedTitle])||([StrFind,idSavedTitle,idCurrentTitle]!=0),[VarSet,idSameTool,0][LoopExit]]
        ,idIdentityIndex]
        [SubToolSelect,idIdentitySelected]
    ]
]
[RoutineDef,ID_Restore,
    [If,[MemGetSize,ID32Recovery]>0,
        [If,[MVarGet,ID32Recovery,0]>0,
            [RoutineCall,ID_CheckIdentity]
            [If,idSameTool==0,
                [RoutineCall,ID_ReadPaths]
                [MessageOK,"ID recovery requires the same Tool and unchanged SubTool names/order. Restore that Tool and press ID again. The backup is in the session Recovery folder.","Generar y exportar ID"]
                [RoutineCall,ID_ShowBackup]
                [Exit]
            ]
            [If,([IsEnabled,"Transform:Edit"]==0)||([IGet,"Transform:Edit"]!=1),
                [MessageOK,"ID recovery is pending. Return the same Tool to 3D Edit mode and press this action again; the recovery marker was kept.","Generar y exportar ID"]
                [Exit]
            ]
            [RoutineCall,ID_ResolvePaths]
            [If,[MVarGet,ID32Recovery,0]<1,
                // Snapshot interruption changed selection/view only; never
                // restore unrecorded per-mesh flags from incomplete entries.
                [SubToolSelect,[MVarGet,ID32Recovery,1]]
                [MTransformSet,ID32Recovery,32]
                [CanvasZoomSet,[MVarGet,ID32Recovery,14]]
                [CanvasPanSet,[MVarGet,ID32Recovery,15],[MVarGet,ID32Recovery,16]]
                [MVarSet,ID32Recovery,0,0]
            ,
            // Keep marker set until the entire restoration succeeds.
            [MVarSet,ID32Recovery,24,1]
            [If,[IExists,"Draw:Lock Camera"],[IUnPress,"Draw:Lock Camera"]]
            [If,[IExists,"Transform:Solo"],[IUnPress,"Transform:Solo"]]
            [Loop,[SubToolGetCount],
                [SubToolSelect,idRestoreIndex]
                [VarSet,idRestoreBase,64+(idRestoreIndex*6)]
                [VarSet,idRestorePP,[MVarGet,ID32Recovery,idRestoreBase+2]]
                [If,(idKeepColorize==1)&&([MVarGet,ID32Recovery,idRestoreBase+4]==1),[VarSet,idRestorePP,1]]
                [If,idRestorePP>=0,
                    [If,[IsEnabled,"Tool:Polypaint:Colorize"],[ISet,"Tool:Polypaint:Colorize",idRestorePP]]
                ]
                [If,[MVarGet,ID32Recovery,idRestoreBase+3]>=0,
                    [If,[IsEnabled,"Tool:Texture Map:Texture On"],[ISet,"Tool:Texture Map:Texture On",[MVarGet,ID32Recovery,idRestoreBase+3]]]
                ]
                [SubToolSetStatus,idRestoreIndex,[MVarGet,ID32Recovery,idRestoreBase+1]]
            ,idRestoreIndex]
            [SubToolSelect,[MVarGet,ID32Recovery,1]]
            [RoutineCall,ID_CB30_Resize,[MVarGet,ID32Recovery,11],[MVarGet,ID32Recovery,12]]
            [ISet,"Document:Pro",[MVarGet,ID32Recovery,13]]
            [RoutineCall,ID_CB30_LensPaths]
            [VarSet,idLens(10),idCamUniversalPath]
            // Universal mode can change the available FOV path. Resolve again
            // after it is restored, with Perspective enabled for focal edits.
            [If,([StrLength,idLens(10)]>0)&&([MVarGet,ID32Recovery,58]>=0),
                [If,[IsEnabled,idLens(10)],[ISet,idLens(10),[MVarGet,ID32Recovery,58]]]
            ]
            [RoutineCall,ID_CB30_LensPaths]
            [VarSet,idLens(9),idCamFovPath]
            [If,[IExists,"Draw:Perspective"],[IPress,"Draw:Perspective"]]
            [Loop,12,
                [VarSet,idLensIndex,idLensRestore]
                [If,((idLensIndex!=0)&&(idLensIndex!=8))&&(idLensIndex!=10),
                    [If,([StrLength,idLens(idLensIndex)]>0)&&([MVarGet,ID32Recovery,48+idLensIndex]>=0),
                        [If,[IsEnabled,idLens(idLensIndex)],[ISet,idLens(idLensIndex),[MVarGet,ID32Recovery,48+idLensIndex]]]
                    ]
                ]
            ,idLensRestore]
            [If,[MVarGet,ID32Recovery,48]>=0,[ISet,"Draw:Perspective",[MVarGet,ID32Recovery,48]]]
            [ISet,"Material:ItemInfo",[MVarGet,ID32Recovery,4]]
            [IColorSet,[MVarGet,ID32Recovery,5],[MVarGet,ID32Recovery,6],[MVarGet,ID32Recovery,7]]
            [IUnPress,idControls(3)][IUnPress,idControls(4)][IUnPress,idControls(5)]
            [If,[MVarGet,ID32Recovery,8]==1,[IPress,idControls(3)]]
            [If,[MVarGet,ID32Recovery,9]==1,[IPress,idControls(4)]]
            [If,[MVarGet,ID32Recovery,10]==1,[IPress,idControls(5)]]
            [ISet,idControls(19),[MVarGet,ID32Recovery,21]]
            [ISet,idPolyPath,[MVarGet,ID32Recovery,19]]
            [If,[MVarGet,ID32Recovery,20]==0,[IPress,"Render:Preview"]]
            [If,[MVarGet,ID32Recovery,20]==1,[IPress,"Render:Best"]]
            [If,[MVarGet,ID32Recovery,20]==2,[IPress,"Render:Fast"]]
            [If,[MVarGet,ID32Recovery,20]==3,[IPress,"Render:Flat"]]
            [If,[IExists,"Transform:Solo"],[ISet,"Transform:Solo",[MVarGet,ID32Recovery,17]]]
            [If,[IExists,"Preferences:LightBox:LightBox"],[ISet,"Preferences:LightBox:LightBox",[MVarGet,ID32Recovery,22]]]
            [If,[IExists,"Preferences:Thumbnail:T.Thumbnail"],[ISet,"Preferences:Thumbnail:T.Thumbnail",[MVarGet,ID32Recovery,23]]]
            [MTransformSet,ID32Recovery,32]
            [CanvasZoomSet,[MVarGet,ID32Recovery,14]]
            [CanvasPanSet,[MVarGet,ID32Recovery,15],[MVarGet,ID32Recovery,16]]
            [If,[MVarGet,ID32Recovery,56]>=0,[ISet,"Draw:Lock Camera",[MVarGet,ID32Recovery,56]]]
            [MVarSet,CB25Data,5,[MVarGet,ID32Recovery,18]]
            [VarSet,idOldToolPath,""]
            [MemReadString,ID32ToolPath,idOldToolPath,0]
            [If,[ToolSetPath,[ToolGetActiveIndex],idOldToolPath]!=0,
                [MessageOK,"ID display recovery could not restore the original Tool save path. The session marker was kept; check the Tool path before saving.","Generar y exportar ID"]
                [Exit]
            ]
            [FileNameSetNext]
            [If,idKeepColorize!=2,[MVarSet,ID32Recovery,0,0]]
            [IUpdate,2,1]
            ]
        ]
    ]
,idKeepColorize]
[RoutineDef,ID_Fail,
    [VarSet,idFailurePhase,0]
    [If,[MemGetSize,ID32Recovery]>0,[VarSet,idFailurePhase,[MVarGet,ID32Recovery,0]]]
    [RoutineCall,ID_ReadPaths]
    [RoutineCall,ID_Restore,0]
    // Keep the error/path separate from mutation status: appending status to a
    // path-bearing error can exceed native ZScript's 255-character strings.
    [MessageOK,idFailureMessage,"Generar y exportar ID"]
    [If,idFailurePhase>=3,
        [MessageOK,"Mesh data may have changed. Load the pre-ID ZTL in Recovery to undo mesh edits. Display recovery only restores flags/view.","Generar y exportar ID"]
    ,
        [MessageOK,"No ID mesh processing started.","Generar y exportar ID"]
    ]
    [RoutineCall,ID_ShowBackup]
    [Exit]
,idFailureMessage]
[RoutineDef,ID_Require,
    [If,[IExists,idRequiredPath]==0,[RoutineCall,ID_Fail,[StrMerge,"Missing control: ",idRequiredPath]]]
    [If,[IsEnabled,idRequiredPath]==0,[RoutineCall,ID_Fail,[StrMerge,"Unavailable control: ",idRequiredPath]]]
,idRequiredPath]
[RoutineDef,ID_ResolvePaths,
    // Old internal names are accepted only when the current named item is absent.
    [If,[IExists,idControls(4)]==0,[VarSet,idControls(4),"Draw:RGB"]]
    [If,[IExists,idControls(5)]==0,[VarSet,idControls(5),"Draw:MRGB"]]
    [If,[IExists,idControls(6)]==0,[VarSet,idControls(6),"Tool:Polygroups:AutoGroups"]]
    [If,[IExists,idControls(19)]==0,[VarSet,idControls(19),"Draw:RGBIntensity"]]
    // ZBrush 2023 exposes the full Draw Polyframe title, not the PolyF label.
    [VarSet,idPolyPath,"Transform:Draw Polyframe"]
    [If,[IExists,idPolyPath]==0,[VarSet,idPolyPath,"Transform:Draw PolyFrame"]]
    [If,[IExists,idPolyPath]==0,[VarSet,idPolyPath,"Transform:pf"]]
    [If,[IExists,idPolyPath]==0,[VarSet,idPolyPath,"Transform:Polyframe"]]
    [If,[IExists,idPolyPath]==0,[VarSet,idPolyPath,"Transform:PolyF"]]
    // Report the canonical path if no alias exists. Presence-based resolution
    // keeps a disabled canonical control from being bypassed through aliases.
    [If,[IExists,idPolyPath]==0,[VarSet,idPolyPath,"Transform:Draw Polyframe"]]
]
[RoutineDef,ID_ResolveControls,
    [RoutineCall,ID_ResolvePaths]
    [RoutineCall,ID_Require,idPolyPath]
    [Loop,20,
        // Mesh-specific controls are checked after selecting every needed mesh.
        [If,(idControlIndex<6)||(idControlIndex>8),
            [If,idControlIndex!=18,
                [If,(idControlIndex>=10)&&(idControlIndex<=15),
                    // Document sizing/redraw is performed with Edit off by the
                    // reviewed reader. These items need not be enabled now.
                    [If,[IExists,idControls(idControlIndex)]==0,[RoutineCall,ID_Fail,[StrMerge,"Missing control: ",idControls(idControlIndex)]]]
                ,
                    [RoutineCall,ID_Require,idControls(idControlIndex)]
                ]
            ]
        ]
    ,idControlIndex]
    [If,[IGet,"Transform:Edit"]!=1,[RoutineCall,ID_Fail,"No live Tool in 3D Edit mode. Draw a PolyMesh3D Tool and press T first."]]
    [If,([SubToolGetCount]<1)||([SubToolGetCount]>1000),[RoutineCall,ID_Fail,"A Tool with 1 to 1000 SubTools is required."]]
]
[RoutineDef,ID_MaterialCandidate,
    [If,[IExists,[Var,idCandidate]],
        // Native 2023 does not reliably evaluate IGetTitle nested in StrFind.
        [VarSet,idMaterialPath,[IGetTitle,[Var,idCandidate],1]]
        [If,[StrFind,"Material:",idMaterialPath]==0,
            [VarSet,idMatName,[IGetTitle,[Var,idCandidate]]]
            [If,([StrLength,idMatName]==10)&&([StrFind,"Flat Color",idMatName]==0),[VarSet,idFlat,idCandidate]]
        ]
    ]
,idCandidate]
[RoutineDef,ID_FindFlat,
    [VarSet,idFlat,-1]
    [If,[IExists,"Material:Flat Color"],[RoutineCall,ID_MaterialCandidate,[IGetID,"Material:Flat Color"]]]
    [If,idFlat<0,
        [IFreeze,[Loop,65536,
            [RoutineCall,ID_MaterialCandidate,idScan]
            [If,idFlat>=0,[LoopExit]]
        ,idScan]]
    ]
    [If,idFlat<0,[RoutineCall,ID_Fail,"Flat Color material is unavailable. Load Flat Color before using ID."]]
    [RoutineCall,ID_Require,[Var,idFlat]]
]
[RoutineDef,ID_CheckBank,
    [If,[MemGetSize,CB25Data]!=idBankBytes,[RoutineCall,ID_Fail,"Load the matching Camera Bank source, then import or store the cameras first."]]
    [If,([MVarGet,CB25Data,0]!=252523)||([MVarGet,CB25Data,1]!=idBankVersion)||([MVarGet,CB25Data,2]!=25),[RoutineCall,ID_Fail,"Unsupported Camera Bank data. Rebuild ID with the matching shared reader."]]
    // Reject legacy positional-only v8 data; identifiers must decide visibility.
    [If,idBankVersion==8,
        [If,[MVarGet,CB25Data,7]!=1,[RoutineCall,ID_Fail,"This bank has positional visibility only. Update cameras with the matching Camera Bank before ID export; name tracking is required."]]
    ]
    [VarSet,idCameraCount,0]
    [Loop,25,
        [VarSet,idBankBase,20+(idCheckSlot*20)]
        [If,([MVarGet,CB25Data,idBankBase]==1)&&([MVarGet,CB25Data,idBankBase+16]==1),
            [VarInc,idCameraCount]
            [If,([MVarGet,CB25Data,idBankBase+18]<=0)||([MVarGet,CB25Data,idBankBase+19]<=0),[RoutineCall,ID_Fail,"An enabled camera has no saved document size. Update it first."]]
            [If,([MVarGet,CB25Data,idBankBase+18]>[IGetMax,"Document:Width"])||([MVarGet,CB25Data,idBankBase+19]>[IGetMax,"Document:Height"]),[RoutineCall,ID_Fail,"A camera exceeds the current document size limits."]]
        ]
    ,idCheckSlot]
    [If,idCameraCount==0,[RoutineCall,ID_Fail,"Enable at least one occupied Camera Bank view (orange). Gray cameras are omitted."]]
]
[RoutineDef,ID_CheckDLL,

    [If,[FileExists,idDll]==0,[RoutineCall,ID_Fail,"ZFileUtils64.dll is missing. Install the ID plugin dependency first."]]
    [If,[FileExecute,idDll,"Version"]<8,[RoutineCall,ID_Fail,"ZFileUtils v8 or newer is required."]]
]

[RoutineDef,ID_ChooseRoot,
    [VarSet,idRoot,""]
    [If,[MemGetSize,RC30Root]>0,[MemReadString,RC30Root,idRoot,0,1]]
    [If,[StrLength,idRoot]>0,
        [If,[FileExecute,idDll,"FolderExists",idRoot]==0,[VarSet,idRoot,""]]
    ]
    [If,[StrLength,idRoot]==0,
        [If,[MemGetSize,ID32Choose]>0,[MemDelete,ID32Choose]]
        [MemCreate,ID32Choose,1024,0]
        [VarSet,idChoice,[FileExecute,idDll,"ChooseFolder",,,ID32Choose]]
        [VarSet,idParent,""][MemReadString,ID32Choose,idParent,0,1]
        [MemDelete,ID32Choose]
        [If,(idChoice!=0)||([StrLength,idParent]==0),[RoutineCall,ID_Fail,"Output selection cancelled."]]
        [If,[StrLength,idParent]>160,[RoutineCall,ID_Fail,"Choose a shorter parent path (<=160 characters)."]]
        [VarSet,idSession,1]
        [Loop,9999,
            [VarSet,idRoot,[StrMerge,idParent,"/ZPass_",idSession]]
            [If,([FileExecute,idDll,"FolderExists",idRoot]==0)&&([FileExists,idRoot]==0),[LoopExit]]
            [VarInc,idSession]
        ]
        [If,idSession>9999,[RoutineCall,ID_Fail,"No free ZPass session name. Choose another parent folder."]]
        [RoutineCall,ID_Folder,idRoot]
    ]
    // MemReadString's supported string limit is 255; leave room for suffixes.
    [If,[StrLength,idRoot]>180,[RoutineCall,ID_Fail,"Output path is too long. Choose a shorter output folder (session path <=180 characters)."]]
]
[RoutineDef,ID_WriteProbe,
    [VarSet,idProbeNumber,1]
    [Loop,9999,
        [VarSet,idProbe,[StrMerge,idProbeFolder,"/ID_write_test_",idProbeNumber,".tmp"]]
        [If,[FileExists,idProbe]==0,[LoopExit]]
        [VarInc,idProbeNumber]
    ]
    [If,idProbeNumber>9999,[RoutineCall,ID_Fail,"Cannot reserve an ID write test filename."]]
    [If,[MemGetSize,ID32Probe]>0,[MemDelete,ID32Probe]]
    [MemCreate,ID32Probe,16,0][MemWriteString,ID32Probe,"ID write check",0]
    [VarSet,idProbeResult,[MemSaveToFile,ID32Probe,idProbe]]
    [MemDelete,ID32Probe]
    [If,idProbeResult!=16,[RoutineCall,ID_Fail,[StrMerge,"Output is not writable: ",idProbeFolder]]]
    [VarSet,idProbeRead,[MemCreateFromFile,ID32Probe,idProbe]]
    [VarSet,idProbeText,""]
    [If,idProbeRead==16,[MemReadString,ID32Probe,idProbeText,0]]
    [If,[MemGetSize,ID32Probe]>0,[MemDelete,ID32Probe]]
    [If,(idProbeRead!=16)||([StrFind,"ID write check",idProbeText]!=0),[RoutineCall,ID_Fail,"Output write/read verification failed."]]
    [If,[FileDelete,idProbe]!=0,[RoutineCall,ID_Fail,"Cannot remove the output write test file."]]
    [If,[FileExists,idProbe]>0,[RoutineCall,ID_Fail,"Output write test cleanup failed."]]
,idProbeFolder]

[RoutineDef,ID_ValidateBackupStem,
    [VarSet,idFilenameError,""]
    [If,[StrLength,idBackupStem]==0,
        [VarSet,idFilenameError,"The first SubTool has an empty filename. No backup was saved."]
    ,
        [Loop,[StrLength,idBackupStem],
            // Pre-evaluate ASCII values; direct StrToAsc conditions have failed
            // in the controller's native ZBrush 2023 parser probes.
            [VarSet,idFilenameCode,[StrToAsc,idBackupStem,idFilenameIndex]]
            [If,(idFilenameCode<32)||(idFilenameCode==127),[VarSet,idFilenameError,"The first SubTool name contains a control character."]]
            [If,((idFilenameCode==34)||(idFilenameCode==42))||((idFilenameCode==47)||(idFilenameCode==58)),[VarSet,idFilenameError,"The first SubTool name contains an unsafe filename character."]]
            [If,((idFilenameCode==60)||(idFilenameCode==62))||((idFilenameCode==63)||(idFilenameCode==92)),[VarSet,idFilenameError,"The first SubTool name contains an unsafe filename character."]]
            [If,idFilenameCode==124,[VarSet,idFilenameError,"The first SubTool name contains an unsafe filename character."]]
            [If,[StrLength,idFilenameError]>0,[LoopExit]]
        ,idFilenameIndex]
        [VarSet,idFilenameCode,[StrToAsc,idBackupStem,[StrLength,idBackupStem]-1]]
        [If,(idFilenameCode==32)||(idFilenameCode==46),[VarSet,idFilenameError,"The first SubTool name ends in a space or period. An exact-name backup is unsafe."]]
        // Windows device names remain reserved even before a file extension.
        [VarSet,idDeviceName,[StrLower,idBackupStem]]
        [VarSet,idFilenameDot,[StrFind,".",idDeviceName]]
        [If,idFilenameDot>0,[VarSet,idDeviceName,[StrExtract,idDeviceName,0,idFilenameDot-1]]]
        [If,[StrLength,idDeviceName]==3,
            [If,([StrFind,"con",idDeviceName]==0)||([StrFind,"prn",idDeviceName]==0),[VarSet,idFilenameError,"The first SubTool name is a reserved Windows filename."]]
            [If,([StrFind,"aux",idDeviceName]==0)||([StrFind,"nul",idDeviceName]==0),[VarSet,idFilenameError,"The first SubTool name is a reserved Windows filename."]]
        ]
        [If,[StrLength,idDeviceName]==4,
            [VarSet,idDevicePrefix,[StrExtract,idDeviceName,0,2]]
            [If,([StrFind,"com",idDevicePrefix]==0)||([StrFind,"lpt",idDevicePrefix]==0),
                [VarSet,idDeviceDigit,[StrToAsc,idDeviceName,3]]
                [If,((idDeviceDigit>=49)&&(idDeviceDigit<=57))||((idDeviceDigit==185)||(idDeviceDigit==178))||((idDeviceDigit==179)),[VarSet,idFilenameError,"The first SubTool name is a reserved Windows filename."]]
            ]
        ]
    ]
]
[RoutineDef,ID_BackupTool,
    // Revert camera-preflight display before the temporary top-SubTool save
    // selection. The backup filename must retain that mesh's original name.
    [RoutineCall,ID_Restore,2]
    [VarSet,idBackupStem,""][MemReadString,ID32Names,idBackupStem,0]
    // IGetTitle adds one UI period. Remove only that final decoration; retain
    // the complete original mesh name, including its camera marker and dots.
    [If,[StrLength,idBackupStem]>0,
        [VarSet,idFilenameLast,[StrToAsc,idBackupStem,[StrLength,idBackupStem]-1]]
        [If,idFilenameLast==46,
            [If,[StrLength,idBackupStem]>1,
                [VarSet,idBackupStem,[StrExtract,idBackupStem,0,[StrLength,idBackupStem]-2]]
            ,
                [VarSet,idBackupStem,""]
            ]
        ]
    ]
    [RoutineCall,ID_ValidateBackupStem]
    [If,[StrLength,idFilenameError]>0,[RoutineCall,ID_Fail,idFilenameError]]
    [VarSet,idBackupNumber,1]
    [Loop,9999,
        [VarSet,idBackupDir,[StrMerge,idRoot,"/Recovery/Before_ID_",idBackupNumber]]
        [If,([FileExists,idBackupDir]==0)&&([FileExecute,idDll,"FolderExists",idBackupDir]==0),
            // Do not reuse names occupied by the former flat-file contract.
            [If,([FileExists,[StrMerge,idBackupDir,".ztl"]]==0)&&([FileExists,[StrMerge,idBackupDir,".txt"]]==0),[LoopExit]]
        ]
        [VarInc,idBackupNumber]
    ]
    [If,idBackupNumber>9999,[RoutineCall,ID_Fail,"No free ID recovery folder."]]
    // Check lengths BEFORE StrMerge. Long names must never be truncated or
    // substituted: that would rename the first SubTool when Save As runs.
    [If,([StrLength,idBackupDir]+[StrLength,idBackupStem]+5)>240,[RoutineCall,ID_Fail,"The exact SubTool name makes the backup path too long (maximum 240). Use a shorter session path. No names were changed."]]
    [VarSet,idBackup,[StrMerge,idBackupDir,"/",idBackupStem,".ztl"]]
    [VarSet,idRecord,[StrMerge,idBackupDir,"/Recovery.txt"]]
    [If,[FileExecute,idDll,"MakeFolder",idBackupDir]!=0,[RoutineCall,ID_Fail,"Cannot reserve a fresh ID recovery folder."]]
    [If,[FileExecute,idDll,"FolderExists",idBackupDir]==0,[RoutineCall,ID_Fail,"Cannot verify the fresh ID recovery folder."]]
    [RoutineCall,ID_WriteProbe,idBackupDir]
    [MemWriteString,ID32BackupPath,idBackup,0][MemWriteString,ID32RecordPath,idRecord,0]
    // Reserve the record without overwrite, then use only this owned record.
    [If,[MemGetSize,ID32Record]>0,[MemDelete,ID32Record]]
    [MemCreate,ID32Record,1024,0]
    [MemWriteString,ID32Record,"Backup requested; no ID mesh processing started.",0]
    [VarSet,idReserve,[MemSaveToFile,ID32Record,idRecord]]
    [MemDelete,ID32Record]
    [If,idReserve!=1024,[RoutineCall,ID_Fail,"Cannot reserve ID recovery record."]]
    // Saving from the top SubTool is the documented SubTool Master convention.
    // Native Save As renames that mesh to the basename. The basename must be
    // its EXACT original name, not the unique recovery folder's label.
    [SubToolSelect,0]
    [RoutineCall,ID_CheckIdentity]
    [If,idSameTool==0,[RoutineCall,ID_Fail,"Tool identity changed before backup. ID processing stopped."]]
    [If,[FileExists,idBackup]>0,[RoutineCall,ID_Fail,"Backup target appeared before save. Existing files were preserved."]]
    [FileNameSetNext,idBackup]
    [IPress,"Tool:Save As"]
    [FileNameSetNext]
    // Check names BEFORE ToolSetPath: it is not a name-repair mechanism.
    [RoutineCall,ID_CheckIdentity]
    [If,idSameTool==0,[RoutineCall,ID_Fail,"Tool names changed during backup. No ID processing started."]]
    [VarSet,idOldToolPath,""][MemReadString,ID32ToolPath,idOldToolPath,0]
    [If,[ToolSetPath,[ToolGetActiveIndex],idOldToolPath]!=0,[RoutineCall,ID_Fail,"Cannot restore the original Tool save path after backup."]]
    [SubToolSelect,[MVarGet,ID32Recovery,1]]
    [RoutineCall,ID_CheckIdentity]
    [If,idSameTool==0,[RoutineCall,ID_Fail,"Tool identity changed after restoring its path. No ID processing started."]]
    [If,([FileExists,idBackup]==0)||([FileGetInfo,idBackup,1]<=0),[RoutineCall,ID_Fail,"ZTL backup failed or was cancelled."]]
    // Verify the file is readable, without loading/replacing the live Tool.
    [If,[MemGetSize,ID32Verify]>0,[MemDelete,ID32Verify]]
    [VarSet,idBackupRead,[MemCreateFromFile,ID32Verify,idBackup,0,64]]
    [If,[MemGetSize,ID32Verify]>0,[MemDelete,ID32Verify]]
    [If,idBackupRead!=64,[RoutineCall,ID_Fail,"ZTL backup is incomplete or unreadable. No ID processing started."]]
    [RoutineCall,ID_CheckIdentity]
    [If,idSameTool==0,[RoutineCall,ID_Fail,"Tool identity changed during backup. ID processing stopped."]]
    [MVarSet,ID32Recovery,0,2]
    [RoutineCall,ID_SaveRecord,"Backup saved/read-checked; ID processing has not started"]
    [NoteBar,[StrMerge,"Backup saved: ",idBackup],0]
]
