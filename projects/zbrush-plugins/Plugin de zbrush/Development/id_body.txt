// ID32Recovery: 0 phase (0.5 partial snapshot, 1 display, 3 mutation), 1 selected,
// 2 count, 3 Tool index, 4 material, 5..7 RGB, 8..10 M/RGB/MRGB,
// 11 width,12 height,13 Pro,14 zoom,15/16 pan,17 Solo,18 active camera,
// 19 PolyFrame,20 render mode,21 RGB intensity,22 LightBox,23 Thumbnail,
// 24 restore in progress,25 completed snapshot entries; 32..40 transform;
// 48..59 optional lens fields;
// 64+i*6: ID, status, Colorize, texture toggle (-1 absent), union, reserved.
// ID32Names uses 256-byte slots. Full titles >255 bytes are rejected.
[RoutineDef,ID_CheckIdentity,
    [VarSet,idSameTool,1]
    [If,([SubToolGetCount]!=[MVarGet,ID32Recovery,2])||([ToolGetActiveIndex]!=[MVarGet,ID32Recovery,3]),[VarSet,idSameTool,0]]
    [If,idSameTool==1,
        [VarSet,idIdentitySelected,[SubToolGetActiveIndex]]
        [VarSet,idIdentityCount,[SubToolGetCount]]
        [If,[MVarGet,ID32Recovery,0]<1,[VarSet,idIdentityCount,[MVarGet,ID32Recovery,25]]]
        [Loop,idIdentityCount,
            [VarSet,idSavedID,0]
            [MemRead,ID32Recovery,idSavedID,6,(64+(idIdentityIndex*6))*4]
            [If,[SubToolGetID,idIdentityIndex]!=idSavedID,[VarSet,idSameTool,0][LoopExit]]
            [SubToolSelect,idIdentityIndex]
            [VarSet,idSavedTitle,""]
            [MemReadString,ID32Names,idSavedTitle,idIdentityIndex*256]
            [VarSet,idCurrentTitle,[IGetTitle,"Tool:ItemInfo"]]
            [If,([StrLength,idCurrentTitle]!=[StrLength,idSavedTitle])||([StrFind,idSavedTitle,idCurrentTitle]!=0),[VarSet,idSameTool,0][LoopExit]]
        ,idIdentityIndex]
        [SubToolSelect,idIdentitySelected]
    ]
]
[RoutineDef,ID_Restore,
    [If,[MemGetSize,ID32Recovery]>0,
        [If,[MVarGet,ID32Recovery,0]>0,
            [If,[MVarGet,ID32Recovery,26]!=325,
                [MessageOK,"A legacy ID export needs recovery with its previous plugin before switching camera providers. The existing recovery record was kept.","Generar y exportar ID"][Exit]
            ]
            [RoutineCall,ID_CheckIdentity]
            [If,idSameTool==0,
                [MessageOK,"ID recovery requires the same Tool and unchanged SubTool names/order. Restore that Tool and press ID again. No automatic Tool backup is created.","Generar y exportar ID"]
                [Exit]
            ]
            [If,([IsEnabled,"Transform:Edit"]==0)||([IGet,"Transform:Edit"]!=1),
                [MessageOK,"ID recovery is pending. Return the same Tool to 3D Edit mode and press this action again; the recovery marker was kept.","Generar y exportar ID"]
                [Exit]
            ]
            [RoutineCall,ID_ResolvePaths]
            [If,[MVarGet,ID32Recovery,0]<1,
                // Snapshot interruption changed selection/view only; never
                // restore unrecorded per-mesh flags from incomplete entries.
                [SubToolSelect,[MVarGet,ID32Recovery,1]]
                [MTransformSet,ID32Recovery,32]
                [CanvasZoomSet,[MVarGet,ID32Recovery,14]]
                [CanvasPanSet,[MVarGet,ID32Recovery,15],[MVarGet,ID32Recovery,16]]
                [MVarSet,ID32Recovery,0,0]
            ,
            // Keep marker set until the entire restoration succeeds.
            [MVarSet,ID32Recovery,24,1]
            [If,[IExists,"Draw:Lock Camera"],[IUnPress,"Draw:Lock Camera"]]
            [If,[IExists,"Transform:Solo"],[IUnPress,"Transform:Solo"]]
            [Loop,[SubToolGetCount],
                [SubToolSelect,idRestoreIndex]
                [VarSet,idRestoreBase,64+(idRestoreIndex*6)]
                [VarSet,idRestorePP,[MVarGet,ID32Recovery,idRestoreBase+2]]
                [If,idRestorePP>=0,
                    [If,[IsEnabled,"Tool:Polypaint:Colorize"],[ISet,"Tool:Polypaint:Colorize",idRestorePP]]
                ]
                [If,[MVarGet,ID32Recovery,idRestoreBase+3]>=0,
                    [If,[IsEnabled,"Tool:Texture Map:Texture On"],[ISet,"Tool:Texture Map:Texture On",[MVarGet,ID32Recovery,idRestoreBase+3]]]
                ]
                [SubToolSetStatus,idRestoreIndex,[MVarGet,ID32Recovery,idRestoreBase+1]]
            ,idRestoreIndex]
            [SubToolSelect,[MVarGet,ID32Recovery,1]]
            [RoutineCall,ID_CB30_Resize,[MVarGet,ID32Recovery,11],[MVarGet,ID32Recovery,12]]
            [ISet,"Document:Pro",[MVarGet,ID32Recovery,13]]
            [RoutineCall,ID_CB30_LensPaths]
            [VarSet,idLens(10),idCamUniversalPath]
            // Universal mode can change the available FOV path. Resolve again
            // after it is restored, with Perspective enabled for focal edits.
            [If,([StrLength,idLens(10)]>0)&&([MVarGet,ID32Recovery,58]>=0),
                [If,[IsEnabled,idLens(10)],[ISet,idLens(10),[MVarGet,ID32Recovery,58]]]
            ]
            [RoutineCall,ID_CB30_LensPaths]
            [VarSet,idLens(9),idCamFovPath]
            [If,[IExists,"Draw:Perspective"],[IPress,"Draw:Perspective"]]
            [Loop,12,
                [VarSet,idLensIndex,idLensRestore]
                [If,((idLensIndex!=0)&&(idLensIndex!=8))&&(idLensIndex!=10),
                    [If,([StrLength,idLens(idLensIndex)]>0)&&([MVarGet,ID32Recovery,48+idLensIndex]>=0),
                        [If,[IsEnabled,idLens(idLensIndex)],[ISet,idLens(idLensIndex),[MVarGet,ID32Recovery,48+idLensIndex]]]
                    ]
                ]
            ,idLensRestore]
            [If,[MVarGet,ID32Recovery,48]>=0,[ISet,"Draw:Perspective",[MVarGet,ID32Recovery,48]]]
            [ISet,"Material:ItemInfo",[MVarGet,ID32Recovery,4]]
            [IColorSet,[MVarGet,ID32Recovery,5],[MVarGet,ID32Recovery,6],[MVarGet,ID32Recovery,7]]
            [IUnPress,idControls(3)][IUnPress,idControls(4)][IUnPress,idControls(5)]
            [If,[MVarGet,ID32Recovery,8]==1,[IPress,idControls(3)]]
            [If,[MVarGet,ID32Recovery,9]==1,[IPress,idControls(4)]]
            [If,[MVarGet,ID32Recovery,10]==1,[IPress,idControls(5)]]
            [ISet,idControls(19),[MVarGet,ID32Recovery,21]]
            [ISet,idPolyPath,[MVarGet,ID32Recovery,19]]
            [If,[MVarGet,ID32Recovery,20]==0,[IPress,"Render:Preview"]]
            [If,[MVarGet,ID32Recovery,20]==1,[IPress,"Render:Best"]]
            [If,[MVarGet,ID32Recovery,20]==2,[IPress,"Render:Fast"]]
            [If,[MVarGet,ID32Recovery,20]==3,[IPress,"Render:Flat"]]
            [If,[IExists,"Transform:Solo"],[ISet,"Transform:Solo",[MVarGet,ID32Recovery,17]]]
            [If,[IExists,"Preferences:LightBox:LightBox"],[ISet,"Preferences:LightBox:LightBox",[MVarGet,ID32Recovery,22]]]
            [If,[IExists,"Preferences:Thumbnail:T.Thumbnail"],[ISet,"Preferences:Thumbnail:T.Thumbnail",[MVarGet,ID32Recovery,23]]]
            [MTransformSet,ID32Recovery,32]
            [CanvasZoomSet,[MVarGet,ID32Recovery,14]]
            [CanvasPanSet,[MVarGet,ID32Recovery,15],[MVarGet,ID32Recovery,16]]
            [If,[MVarGet,ID32Recovery,56]>=0,[ISet,"Draw:Lock Camera",[MVarGet,ID32Recovery,56]]]
            [If,[MemGetSize,FB25Data]==idBankBytes,
                [If,([MVarGet,FB25Data,8]==[MVarGet,ID32Recovery,27])&&([MVarGet,FB25Data,9]==[MVarGet,ID32Recovery,28]),
                    [MVarSet,FB25Data,5,[MVarGet,ID32Recovery,18]]
                ]
            ]
            [VarSet,idOldToolPath,""]
            [MemReadString,ID32ToolPath,idOldToolPath,0]
            [VarSet,idPathToolIndex,[ToolGetActiveIndex]]
            [VarSet,idPathResult,[ToolSetPath,idPathToolIndex,idOldToolPath]]
            [If,idPathResult!=0,
                [MessageOK,"ID display recovery could not restore the original Tool save path. The session marker was kept; check the Tool path before saving.","Generar y exportar ID"]
                [Exit]
            ]
            [FileNameSetNext]
            [If,idKeepColorize!=2,[MVarSet,ID32Recovery,0,0]]
            [IUpdate,2,1]
            ]
        ]
    ]
,idKeepColorize]

[RoutineDef,ID_ResolveControls,
    [RoutineCall,ID_ResolvePaths]
    [RoutineCall,ID_Require,idPolyPath]
    [Loop,20,
        // Mesh-specific controls are checked after selecting every needed mesh.
        [If,(idControlIndex<6)||(idControlIndex>8),
            [If,(idControlIndex!=18)&&((idExportOnly==0)||(idControlIndex!=2)),
                [If,(idControlIndex>=10)&&(idControlIndex<=15),
                    // Document sizing/redraw is performed with Edit off by the
                    // reviewed reader. These items need not be enabled now.
                    [If,[IExists,idControls(idControlIndex)]==0,[RoutineCall,ID_Fail,[StrMerge,"Missing control: ",idControls(idControlIndex)]]]
                ,
                    [RoutineCall,ID_Require,idControls(idControlIndex)]
                ]
            ]
        ]
    ,idControlIndex]
    [If,idExportOnly==1,[RoutineCall,ID_Require,"Render:Flat"]]
    [If,[IGet,"Transform:Edit"]!=1,[RoutineCall,ID_Fail,"No live Tool in 3D Edit mode. Draw a PolyMesh3D Tool and press T first."]]
    [If,([SubToolGetCount]<1)||([SubToolGetCount]>1000),[RoutineCall,ID_Fail,"A Tool with 1 to 1000 SubTools is required."]]
]

        ]

[RoutineDef,ID_CheckBank,
    [If,[MemGetSize,FB25Data]!=idBankBytes,[RoutineCall,ID_Fail,"Open Camera Bank Fast, then import or store your cameras. Its standalone plugin must be loaded."]]

    [If,idBankVersion>=8,
        [If,[MVarGet,FB25Data,7]!=1,[RoutineCall,ID_Fail,"This bank has positional visibility only. Update cameras with the matching Camera Bank Fast before ID export; name tracking is required."]]
    ]
    [VarSet,idCameraCount,0]
    [Loop,25,
        [VarSet,idBankBase,20+(idCheckSlot*20)]

[RoutineDef,ID_ChooseRoot,
    [VarSet,idRoot,""]
    [If,[MemGetSize,RC30Root]>0,[MemReadString,RC30Root,idRoot,0,1]]
    // Discard pre-3.2.2 session roots; only the fixed output folder is valid.
    [If,[StrLength,idRoot]>0,
        [If,[StrLength,idRoot]<8,[VarSet,idRoot,""],
            [VarSet,idTail,[StrLower,[StrExtract,idRoot,[StrLength,idRoot]-8,[StrLength,idRoot]-1]]]
            [If,([StrFind,"/zpass_1",idTail]!=0)&&([StrFind,"\zpass_1",idTail]!=0),[VarSet,idRoot,""]]
        ]
    ]
    [If,[StrLength,idRoot]>0,
        [If,[FileExecute,idDll,"FolderExists",idRoot]==0,[VarSet,idRoot,""]]
    ]
    [If,[StrLength,idRoot]==0,
        [If,[MemGetSize,ID32Choose]>0,[MemDelete,ID32Choose]]
        [MemCreate,ID32Choose,1024,0]
        [VarSet,idChoice,[FileExecute,idDll,"ChooseFolder",,,ID32Choose]]
        [VarSet,idParent,""][MemReadString,ID32Choose,idParent,0,1]
        [MemDelete,ID32Choose]
        [If,(idChoice!=0)||([StrLength,idParent]==0),[RoutineCall,ID_Fail,"Output selection cancelled."]]
        [If,[StrLength,idParent]>160,[RoutineCall,ID_Fail,"Choose a shorter parent path (<=160 characters)."]]
    // Accept either the parent directory or ZPass_1 itself, including a trailing separator.
    [Loop,1024,
        [If,[StrLength,idParent]<2,[LoopExit]]
        [VarSet,idLast,[StrToAsc,idParent,[StrLength,idParent]-1]]
        [If,(idLast!=47)&&(idLast!=92),[LoopExit]]
        [VarSet,idParent,[StrExtract,idParent,0,[StrLength,idParent]-2]]
    ]
    [VarSet,idRoot,[StrMerge,idParent,"/ZPass_1"]]
    [If,[StrLength,idParent]>=7,
        [VarSet,idTail,[StrLower,[StrExtract,idParent,[StrLength,idParent]-7,[StrLength,idParent]-1]]]
        [If,[StrFind,"zpass_1",idTail]==0,
            [If,[StrLength,idParent]==7,[VarSet,idRoot,idParent],
                [VarSet,idLast,[StrToAsc,idParent,[StrLength,idParent]-8]]
                [If,(idLast==47)||(idLast==92),[VarSet,idRoot,idParent]]
            ]
        ]
    ]
        [RoutineCall,ID_Folder,idRoot]
    ]
    // MemReadString's supported string limit is 255; leave room for suffixes.
    [If,[StrLength,idRoot]>180,[RoutineCall,ID_Fail,"Output path is too long. Choose a shorter output folder (session path <=180 characters)."]]
]

    [If,[IGet,"Render:Best"]==1,[MVarSet,ID32Recovery,20,1]]
    [If,[IGet,"Render:Fast"]==1,[MVarSet,ID32Recovery,20,2]]
    [If,[IGet,"Render:Flat"]==1,[MVarSet,ID32Recovery,20,3]]
    [MVarSet,ID32Recovery,21,[IGet,idControls(19)]]
    [If,[IExists,"Preferences:LightBox:LightBox"],[MVarSet,ID32Recovery,22,[IGet,"Preferences:LightBox:LightBox"]]]
    [If,[IExists,"Preferences:Thumbnail:T.Thumbnail"],[MVarSet,ID32Recovery,23,[IGet,"Preferences:Thumbnail:T.Thumbnail"]]]
    [MTransformGet,ID32Recovery,32]
    [RoutineCall,ID_CB30_LensPaths]
    [VarSet,idLens(9),idCamFovPath][VarSet,idLens(10),idCamUniversalPath]
    [Loop,12,
        [MVarSet,ID32Recovery,48+idLensSave,-1]
        [If,[StrLength,idLens(idLensSave)]>0,
            [If,[IExists,idLens(idLensSave)],[MVarSet,ID32Recovery,48+idLensSave,[IGet,idLens(idLensSave)]]]
        ]
    ,idLensSave]
    // A partial snapshot can recover selection/view without guessing flags.
    [MVarSet,ID32Recovery,0,0.5]
    [VarSet,idNameError,0]
    [Loop,[SubToolGetCount],
        [SubToolSelect,idSnapshotIndex]
        [VarSet,idName,[IGetTitle,"Tool:ItemInfo"]]
        [If,[StrLength,idName]>255,[VarSet,idNameError,1][LoopExit]]
        [MemWriteString,ID32Names,idName,idSnapshotIndex*256]
        [VarSet,idSnapshotBase,64+(idSnapshotIndex*6)]
        // Native IDs are uint32; float cells would round values above 2^24.
        [MemWrite,ID32Recovery,[SubToolGetID,idSnapshotIndex],6,idSnapshotBase*4]
        [MVarSet,ID32Recovery,idSnapshotBase+1,[SubToolGetStatus,idSnapshotIndex]]
        [MVarSet,ID32Recovery,idSnapshotBase+2,-1]
        [If,[IExists,"Tool:Polypaint:Colorize"],[MVarSet,ID32Recovery,idSnapshotBase+2,[IGet,"Tool:Polypaint:Colorize"]]]
        [MVarSet,ID32Recovery,idSnapshotBase+3,-1]
        [If,[IExists,"Tool:Texture Map:Texture On"],[MVarSet,ID32Recovery,idSnapshotBase+3,[IGet,"Tool:Texture Map:Texture On"]]]
        [VarSet,idNeeded(idSnapshotIndex),0]
        [MVarSet,ID32Recovery,25,idSnapshotIndex+1]
    ,idSnapshotIndex]
    [SubToolSelect,[MVarGet,ID32Recovery,1]]
    [MTransformSet,ID32Recovery,32][CanvasZoomSet,[MVarGet,ID32Recovery,14]]
    [CanvasPanSet,[MVarGet,ID32Recovery,15],[MVarGet,ID32Recovery,16]]
    [If,idNameError==1,[RoutineCall,ID_Fail,"SubTool name exceeds the safe recovery title length."]]
    [MVarSet,ID32Recovery,0,1]
]
[RoutineDef,ID_CollectUnion,
    // Recall through the shared name-aware reader, then inspect effective eyes.
    [Loop,25,
        [VarSet,idUnionBase,20+(idUnionSlot*20)]
        [If,([MVarGet,FB25Data,idUnionBase]==1)&&([MVarGet,FB25Data,idUnionBase+16]==1),
            [RoutineCall,ID_CB32_RequireVisibility,idUnionSlot]
            [RoutineCall,ID_CB29_LoadSlot,idUnionSlot]
            [VarSet,idVisibleCount,0]
            [Loop,[SubToolGetCount],
                [RoutineCall,ID_CB32_VisibleNow,idUnionIndex]
                [If,idCamTrackVisible==1,[VarSet,idNeeded(idUnionIndex),1][VarInc,idVisibleCount]]
            ,idUnionIndex]
            [If,idVisibleCount==0,[RoutineCall,ID_Fail,"An enabled camera has no effective visible mesh."]]
        ]
    ,idUnionSlot]
    [VarSet,idProcessedCount,0]
    [Loop,[SubToolGetCount],
        [If,idNeeded(idMeshIndex)==1,
            [SubToolSelect,idMeshIndex]
            [RoutineCall,ID_Require,idControls(8)]
            [If,idExportOnly==0,
            [RoutineCall,ID_Require,idControls(6)]
            [RoutineCall,ID_Require,idControls(7)]
            [RoutineCall,ID_Require,idControls(18)]
            [RoutineCall,ID_Require,idControls(2)]
            [If,[IExists,"Tool:Texture Map:Texture On"],
                [If,[IGet,"Tool:Texture Map:Texture On"]==1,[RoutineCall,ID_Require,"Tool:Texture Map:Texture On"]]
            ]
            ]
            [MVarSet,ID32Recovery,64+(idMeshIndex*6)+4,1]
            [VarInc,idProcessedCount]
        ]
    ,idMeshIndex]
]
[RoutineDef,ID_FillMaterialOnly,
    [IPress,[Var,idFlat]]
    [IUnPress,idControls(5)][IUnPress,idControls(4)][IPress,idControls(3)]
    [If,([IGet,idControls(3)]!=1)||([IGet,idControls(4)]!=0)||([IGet,idControls(5)]!=0),[RoutineCall,ID_Fail,"Cannot activate material-only fill mode."]]
    [ISet,idControls(19),100]
    [IPress,idControls(2)]
]
[RoutineDef,ID_PrepareMeshes,
    // Session marker precedes the FIRST mesh operation.
    [MVarSet,ID32Recovery,0,3]
    [If,[IExists,"Transform:Solo"],[IUnPress,"Transform:Solo"]]
    [Loop,[SubToolGetCount],
        [If,idNeeded(idPrepareIndex)==1,
            [SubToolSelect,idPrepareIndex]
            [IPress,idControls(18)]
            [RoutineCall,ID_FillMaterialOnly]
            [IPress,idControls(6)]
            [IPress,idControls(7)]
            [ISet,idControls(8),1]
            [If,[IExists,"Tool:Texture Map:Texture On"],[IUnPress,"Tool:Texture Map:Texture On"]]
            [If,[IGet,idControls(8)]!=1,[RoutineCall,ID_Fail,"Colorize could not be enabled on a required SubTool."]]
        ]
    ,idPrepareIndex]
]
[RoutineDef,ID_EnablePolypaint,
    // Enable existing RGB for every mesh used by the enabled cameras.
    // Run BEFORE camera recall so selection cannot disturb saved framing.
    [Loop,[SubToolGetCount],
        [If,idNeeded(idPaintIndex)==1,
            [SubToolSelect,idPaintIndex]
            [RoutineCall,ID_Require,idControls(8)]
            [ISet,idControls(8),1]
            [If,[IGet,idControls(8)]!=1,[RoutineCall,ID_Fail,"Colorize could not be enabled on a required SubTool."]]
        ]
    ,idPaintIndex]
]
[RoutineDef,ID_ExportCameras,
    [VarSet,idExportedCount,0]
    [Loop,25,
        [VarSet,idExportBase,20+(idExportSlot*20)]
        [If,([MVarGet,FB25Data,idExportBase]==1)&&([MVarGet,FB25Data,idExportBase+16]==1),
            [RoutineCall,ID_CheckIdentity]
            [If,idSameTool==0,[RoutineCall,ID_Fail,"Tool identity changed during ID capture."]]
            [RoutineCall,ID_EnablePolypaint]
            [RoutineCall,ID_CB29_LoadSlot,idExportSlot]
            [If,idExportOnly==0,[IPress,[Var,idFlat]][IColorSet,255,255,255]]
            [IUnPress,idPolyPath]
            [If,[IExists,"Draw:Floor"],[IUnPress,"Draw:Floor"]]
            // Flat rendering captures manual colors/textures without filling
            // materials or changing RGB, PolyGroups or masks. Colorize is on.
            [If,idExportOnly==1,[IPress,"Render:Flat"],[IPress,"Render:Preview"]]
            [IUpdate,2,1]
            [RoutineCall,ID_Label,idExportSlot+1]
            [VarSet,idOutput,[StrMerge,idRoot,"/",idLabel,"/ID.png"]]
            [If,[FileExists,idOutput]>0,
                [If,[FileDelete,idOutput]!=0,[RoutineCall,ID_Fail,[StrMerge,"Cannot replace: ",idOutput]]]
            ]
            [FileNameSetNext,idOutput][IPress,"Document:Export"][FileNameSetNext]
            [If,([FileExists,idOutput]==0)||([FileGetInfo,idOutput,1]<=0),[RoutineCall,ID_Fail,[StrMerge,"ID export failed or was cancelled: ",idOutput]]]
            [If,[MemGetSize,ID32Verify]>0,[MemDelete,ID32Verify]]
            [VarSet,idPNGRead,[MemCreateFromFile,ID32Verify,idOutput,0,8]]
            [If,idPNGRead==8,
                [Loop,8,
                    [VarSet,idPNGByte,0]
                    [MemRead,ID32Verify,idPNGByte,2,idByteIndex]
                    [If,idPNGByte!=idPNG(idByteIndex),[VarSet,idPNGRead,0]]
                ,idByteIndex]
            ]
            [If,[MemGetSize,ID32Verify]>0,[MemDelete,ID32Verify]]
            [If,idPNGRead!=8,[RoutineCall,ID_Fail,[StrMerge,"Export is not a readable PNG: ",idOutput]]]
            [VarInc,idExportedCount]
            [NoteBar,[StrMerge,"ID | Camera ",idLabel],idExportedCount/idCameraCount]
        ]
    ,idExportSlot]
]
[RoutineDef,ID_ShowResult,
    // Max counters (25 / 1000) and root length 180 produce 239 characters.
    [If,idExportOnly==1,
    [MessageOK,[StrMerge,"ID exported: ",idExportedCount," cameras. Output: ",idRoot],"Generar y exportar ID"]
    ,
    [MessageOK,[StrMerge,"ID complete: ",idExportedCount," cameras, ",idProcessedCount," meshes. Output: ",idRoot],"Generar y exportar ID"]
    ]
]
[RoutineDef,ID_Run,
    [If,[MemGetSize,ID32Recovery]>0,
        [If,[MVarGet,ID32Recovery,0]>0,
            [VarSet,idInterruptedPhase,[MVarGet,ID32Recovery,0]]
            [RoutineCall,ID_Restore,0]
            [If,idInterruptedPhase>=3,
                [MessageOK,"Interrupted ID: display/view restored. Mesh edits and exported files remain. No automatic Tool backup was created. Nothing was rerun.","Generar y exportar ID"]
            ,
                [MessageOK,"Interrupted preparation: display/view restored. No ID mesh processing started. Press ID again to start. No automatic Tool backup is created.","Generar y exportar ID"]
            ]
            [Exit]
        ]
    ]
    [VarSet,idExportOnly,idOnly]
    [RoutineCall,ID_ResolveControls]
    [RoutineCall,ID_CheckBank]
    [If,idExportOnly==0,[RoutineCall,ID_FindFlat]]
    [RoutineCall,ID_CheckDLL]
    [RoutineCall,ID_ChooseRoot]
    [RoutineCall,ID_CheckOutput]
    [RoutineCall,ID_Snapshot]
    [RoutineCall,ID_CollectUnion]
    [RoutineCall,ID_Restore,2]
    [If,idExportOnly==0,[RoutineCall,ID_PrepareMeshes]]
    [RoutineCall,ID_ExportCameras]
    [RoutineCall,ID_Restore,1]
    // Publish a new session for the other passes only after success.
    [If,[MemGetSize,RC30Root]>0,[MemDelete,RC30Root]]
    [MemCreate,RC30Root,1024,0][MemWriteString,RC30Root,idRoot,0]
    [RoutineCall,ID_ShowResult]
,idOnly]
